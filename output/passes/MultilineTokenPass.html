Multiline Token Pass

We split token that span multiple lines into multiple tokens where each token are ensured to be in the same line.

import type { TokenInfo } from "../types";
import type { TokenInfoPass } from "./TokenInfoPass";

export class MultilineTokenPass implements TokenInfoPass {
	process(tokens: TokenInfo[]): TokenInfo[] {
		for (const token of tokens) {
			if (token.span.start.line === token.span.end.line) continue;
		}

        // Create new tokens
        const splitedTokens: TokenInfo[] = [];

        // Replace token with new tokens
		return tokens;
	}
}
